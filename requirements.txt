"""
Lumina: FastAPI-обёртка над GraphRAG-пайплайном.

Локальный запуск: uvicorn main:app --reload --port 8000

Эндпоинты:
    GET  /              проверка, что сервис жив
    GET  /api/health    задан ли ключ и какие модели используются
    POST /api/analyze   текст и вопрос на входе, результат пайплайна на выходе
    POST /api/extract   достаёт текст из загруженного PDF
"""

import io
import os
from fastapi import FastAPI, HTTPException, UploadFile, File
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel, Field

import core

app = FastAPI(
    title="Lum / Lumina API",
    description="GraphRAG backend: текст + вопрос → граф знаний + объяснимый ответ",
    version="1.0.0",
)

# CORS: домены фронта берём из ALLOWED_ORIGINS через запятую,
# например "https://lumina.uz,https://lum.pages.dev".
# Без переменной открыто для всех ("*"), для локальной разработки так удобно.
_origins_env = os.environ.get("ALLOWED_ORIGINS", "*")
_allowed_origins = ["*"] if _origins_env.strip() == "*" else [
    o.strip() for o in _origins_env.split(",") if o.strip()
]

app.add_middleware(
    CORSMiddleware,
    allow_origins=_allowed_origins,
    allow_credentials=False,
    allow_methods=["GET", "POST", "OPTIONS"],
    allow_headers=["*"],
)


# схемы запросов

class AnalyzeRequest(BaseModel):
    text: str = Field(..., min_length=1, description="Текст для анализа")
    query: str = Field(..., min_length=1, description="Вопрос пользователя")


# максимальный размер PDF в байтах. PDF разбираем на бэке, так надёжнее, чем в браузере:
# фронт получает чистый текст и шлёт его в /api/analyze
MAX_PDF_BYTES = int(os.environ.get("MAX_PDF_BYTES", str(15 * 1024 * 1024)))  # 15 МБ


# эндпоинты

@app.get("/")
def root():
    return {"status": "ok", "service": "lum-api"}


@app.get("/api/health")
def health():
    return {
        "status": "ok",
        "gemini_key_set": bool(core.GROQ_API_KEY),
        "light_model": core.LIGHT_MODEL,
        "power_model": core.POWER_MODEL,
        "chunk_size": core.CHUNK_SIZE,
        "top_k": core.TOP_K,
    }


@app.post("/api/analyze")
def analyze(req: AnalyzeRequest):
    """Прогоняет текст и вопрос через GraphRAG. Что лежит в ответе, описано в core.run_pipeline."""
    try:
        return core.run_pipeline(text=req.text, query=req.query)
    except core.PipelineError as e:
        # ожидаемые ошибки (пустой ввод, нет ключа, Gemini упал) отдаём как 400
        raise HTTPException(status_code=400, detail=str(e))
    except Exception as e:
        # всё остальное отдаём как 500, подробности наружу не показываем
        raise HTTPException(status_code=500, detail=f"Внутренняя ошибка: {type(e).__name__}")


@app.post("/api/extract")
async def extract_pdf(file: UploadFile = File(...)):
    """
    Принимает PDF и возвращает текст: {text, chars, pages}.
    Дальше фронт шлёт этот text в /api/analyze.
    """
    filename = (file.filename or "").lower()
    is_pdf = filename.endswith(".pdf") or file.content_type == "application/pdf"
    if not is_pdf:
        raise HTTPException(status_code=400, detail="Ожидается PDF-файл (.pdf).")

    raw = await file.read()
    if not raw:
        raise HTTPException(status_code=400, detail="Пустой файл.")
    if len(raw) > MAX_PDF_BYTES:
        mb = MAX_PDF_BYTES // (1024 * 1024)
        raise HTTPException(status_code=400, detail=f"Файл слишком большой (макс. {mb} МБ).")

    try:
        from pypdf import PdfReader
        reader = PdfReader(io.BytesIO(raw))
        # пробуем открыть с пустым паролем, если не вышло, говорим об этом
        if reader.is_encrypted:
            try:
                reader.decrypt("")
            except Exception:
                raise HTTPException(status_code=400, detail="PDF защищён паролем — снимите защиту.")
        pages = len(reader.pages)
        parts = []
        for page in reader.pages:
            chunk = page.extract_text() or ""
            if chunk.strip():
                parts.append(chunk)
        text = "\n\n".join(parts).strip()
    except HTTPException:
        raise
    except Exception as e:
        raise HTTPException(status_code=400, detail=f"Не удалось разобрать PDF: {type(e).__name__}")

    if not text:
        raise HTTPException(
            status_code=400,
            detail="Из PDF не удалось извлечь текст — вероятно, это скан без текстового слоя (нужен OCR).",
        )

    return {"text": text, "chars": len(text), "pages": pages}
